  <RuleShowcase
    number="03"
    id="grouped-signatures"
    title="Grouped signatures"
    description="/ and * are semantic block delimiters. Parameters pack within each block; long names still wrap by width."

  <RuleShowcase
    number="04"
    id="rectangular-data"
    title="Rectangular data"
    description="Extra whitespace that lines up columns is treated as alignment intent. Cells format normally, then column widths and decimal points are enforced across the matrix."
    beforeLabel="Input intent"
    before={String.raw`matrix = [

]`}
  />

  <RuleShowcase
    number="05"
    id="container-hug"
    title="Nested container hug"
    description="When a call’s only argument is a multiline list, tuple, set, or dict, Tali hugs the delimiters instead of adding another indentation layer."
    before={String.raw`render(
    [
        1,
        2,
        3,
    ]
)

    }
)`}
    after={String.raw`render([
    1,
    2,
    3,
])

    "enabled": True,
    "options": ["a", "b"],
})`}
  />

  <RuleShowcase
    number="06"
    id="compact-parens"
    title="Compact parentheses"
    description="Prefer an expression’s own split points before wrapping it in stand-alone parenthesis lines. When parens are needed, keep them beside the expression or use a compact hanging layout."
    before={String.raw`total = (
    first_component
    + second_component
    + third_component
)

value_with_a_long_target = (
    first_component
    + second_component
)`}
    after={String.raw`total = (first_component
    + second_component
    + third_component)

value_with_a_long_target = (
    first_component
    + second_component)`}
  />

  <RuleShowcase
    number="07"
    id="horizontal-align"
    title="Horizontal alignment"
    description="Extra source spacing opts consecutive related statements into alignment: = operators, annotation content after :, and trailing comments when a wide gap signals intent."
    beforeLabel="Input intent"
    before={String.raw`short   = 1
long_name = 2

    after={String.raw`short     = 1
long_name = 2

x:    int       = 1
long: str       = 2

a = 1          # first
long_name = 2  # second`}
  />

  <RuleShowcase
    number="08"
    id="pointer-comments"
    title="Pointer comments"
    description="Trailing comments never force the code before them to wrap. If the full line exceeds the width, the comment drops below as a pointer. Pragmas stay attached."
    beforeLabel="Input"
    before={String.raw`result = some_function(argument)  # This deliberately long explanation makes the complete source line exceed the configured formatter width.

    after={String.raw`result = some_function(argument)
# ^ This deliberately long explanation makes the complete source line exceed the configured formatter width.

pragma_result = some_function(argument)  # noqa: E501`}
  />
